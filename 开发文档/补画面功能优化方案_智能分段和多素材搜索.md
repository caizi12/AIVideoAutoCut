# 补画面功能优化方案 - 智能分段和多素材搜索

**创建日期**: 2026-05-28
**优化目标**:
1. 智能字幕分段，避免机械式6秒分段
2. 每个镜头搜索多个素材候选，提供更多选择

---

## 一、当前问题分析

### 问题1: 机械式分段不合理 ❌

**当前逻辑** (`_split_long_subtitle` 方法):
```python
def _split_long_subtitle(self, item: Dict) -> List[Dict]:
    duration = item['end'] - item['start']
    if duration <= 8.0:
        return [item]
    pieces = max(1, int(duration // 6) + 1)  # 机械式按6秒分段
    piece_duration = duration / pieces
    return [{
        'start': item['start'] + index * piece_duration,
        'end': item['start'] + (index + 1) * piece_duration,
        'text': item['text']
    } for index in range(pieces)]
```

**问题**:
- ❌ 固定6秒分段，不考虑语义
- ❌ 可能在句子中间切断
- ❌ 导致同一段话搜索多次相同素材
- ❌ 不考虑字幕内容的主题变化

**示例**:
```
字幕: "今天我们来介绍一下人工智能的发展历史，从1950年代开始，科学家们就开始研究这个领域" (15秒)
当前分段:
  - 0-6s: "今天我们来介绍一下人工智能的发展历史，从1950年代开始，科学家们就开始研究这个领域"
  - 6-12s: "今天我们来介绍一下人工智能的发展历史，从1950年代开始，科学家们就开始研究这个领域"
  - 12-15s: "今天我们来介绍一下人工智能的发展历史，从1950年代开始，科学家们就开始研究这个领域"

问题: 三个镜头搜索相同关键词 "人工智能、发展、历史"，重复素材
```

## 二、优化方案

### 方案A: AI智能分段（推荐）✅

#### A1. 基于大语言模型的语义分段

**核心思路**: 使用AI理解字幕内容，按语义主题分段

**实现步骤**:

1. **调用AI分析字幕**
```python
def _ai_segment_subtitles(self, subtitles: List[Dict], ai_config: Dict) -> List[Dict]:
    """使用AI智能分段字幕"""

    # 1. 合并所有字幕文本
    full_text = ' '.join([s['text'] for s in subtitles])

    # 2. 构建AI提示词
    prompt = f"""
请分析以下视频字幕，将其分成若干个语义连贯的镜头段落。

字幕内容：
{full_text}

字幕时间轴：
{self._format_subtitle_timeline(subtitles)}

要求：
1. 每个镜头段落应该是一个完整的语义单元（一个话题、一个场景、一个观点）
2. 每个镜头时长建议在3-10秒之间
3. 避免在句子中间切断
4. 为每个镜头提取3-5个关键词，用于搜索视频素材
5. 为每个镜头生成一个简短的画面描述（用于素材搜索）

请以JSON格式返回：
{{
  "segments": [
    {{
      "start_index": 0,  // 起始字幕索引
      "end_index": 2,    // 结束字幕索引
      "theme": "介绍主题",  // 段落主题
      "keywords": ["人工智能", "介绍", "主题"],
      "scene_description": "科技感的城市或实验室场景",
      "search_queries": ["artificial intelligence lab", "technology introduction"]
    }},
    ...
  ]
}}
"""

    # 3. 调用AI API
    ai_response = self._call_ai_api(prompt, ai_config)

    # 4. 解析AI返回的分段结果
    segments = self._parse_ai_segments(ai_response, subtitles)

    return segments
```

**优点**:
- ✅ 语义连贯，不会在句子中间切断
- ✅ 自动识别主题变化
- ✅ 生成更精准的搜索关键词
- ✅ 可以理解上下文，避免重复

**缺点**:
- ⚠️ 需要调用AI API（成本）
- ⚠️ 响应时间较长（2-5秒）
- ⚠️ 依赖AI服务稳定性

#### A2. 基于NLP的规则分段（轻量级）

**核心思路**: 使用NLP技术识别句子边界和主题变化

**实现步骤**:

```python
def _nlp_segment_subtitles(self, subtitles: List[Dict]) -> List[Dict]:
    """使用NLP规则智能分段"""

    segments = []
    current_segment = None

    for i, subtitle in enumerate(subtitles):
        text = subtitle['text']

        # 1. 检测句子结束标点
        is_sentence_end = self._is_sentence_end(text)

        # 2. 检测主题变化（关键词变化）
        keywords = self._extract_keywords(text, '')
        is_topic_change = self._is_topic_change(current_segment, keywords)

        # 3. 检测时长
        if current_segment:
            duration = subtitle['end'] - current_segment['start']
        else:
            duration = 0

        # 4. 决定是否分段
        should_split = (
            is_sentence_end and duration >= 3.0 or  # 句子结束且>=3秒
            is_topic_change or                       # 主题变化
            duration >= 10.0                         # 超过10秒强制分段
        )

        if current_segment is None:
            current_segment = {
                'start': subtitle['start'],
                'end': subtitle['end'],
                'text': text,
                'keywords': keywords
            }
        elif should_split:
            segments.append(current_segment)
            current_segment = {
                'start': subtitle['start'],
                'end': subtitle['end'],
                'text': text,
                'keywords': keywords
            }
        else:
            # 合并到当前段落
            current_segment['end'] = subtitle['end']
            current_segment['text'] += ' ' + text
            current_segment['keywords'] = list(set(current_segment['keywords'] + keywords))

    if current_segment:
        segments.append(current_segment)

    return segments

def _is_sentence_end(self, text: str) -> bool:
    """检测是否句子结束"""
    return bool(re.search(r'[。！？\.!?]$', text.strip()))

def _is_topic_change(self, current_segment: Dict, new_keywords: List[str]) -> bool:
    """检测主题是否变化"""
    if not current_segment:
        return False

    old_keywords = set(current_segment.get('keywords', []))
    new_keywords_set = set(new_keywords)

    # 如果关键词重叠度低于30%，认为主题变化
    if not old_keywords:
        return False

    overlap = len(old_keywords & new_keywords_set)
    overlap_ratio = overlap / len(old_keywords)

    return overlap_ratio < 0.3
```

**优点**:
- ✅ 无需调用外部API
- ✅ 响应快速
- ✅ 成本低
- ✅ 考虑了句子边界和主题变化

**缺点**:
- ⚠️ 准确度不如AI
- ⚠️ 规则可能需要调优

#### A3. 混合方案（推荐）✅

**核心思路**: 优先使用NLP规则，可选启用AI增强

```python
def _build_shots(self, subtitles: List[Dict], data: Dict) -> List[Dict]:
    """生成镜头计划（智能分段版本）"""

    # 1. 检查是否启用AI分段
    use_ai_segment = data.get('use_ai_segment', False)
    ai_config = self._get_ai_config()

    # 2. 智能分段
    if use_ai_segment and ai_config.get('enabled'):
        # 使用AI分段
        segments = self._ai_segment_subtitles(subtitles, ai_config)
    else:
        # 使用NLP规则分段
        merged = self._merge_short_subtitles(subtitles)
        segments = self._nlp_segment_subtitles(merged)

    # 3. 生成镜头列表
    shots = []
    for index, segment in enumerate(segments, start=1):
        keywords = segment.get('keywords', [])
        search_queries = self._build_search_queries(
            keywords,
            segment.get('text', ''),
            data.get('topic', '')
        )

        shots.append({
            'shot_id': f's{index:03d}',
            'start': segment['start'],
            'end': segment['end'],
            'duration': round(segment['end'] - segment['start'], 3),
            'subtitle_text': segment['text'],
            'theme': segment.get('theme', ''),  # AI提供的主题
            'scene_description': segment.get('scene_description', ''),  # AI提供的场景描述
            'keywords': keywords,
            'search_queries': search_queries,
            'locked': False,
            'skipped': False,
            'selected_candidate_id': '',
            'candidates': []
        })

    return shots
```

### 方案B: 多素材搜索优化 ✅

#### B1. 为每个镜头搜索多样化素材

**核心思路**: 确保每个镜头有1-3个不同风格的素材可选

```python
def _search_candidates_for_shot(
    self,
    shot: Dict,
    providers: List,
    orientation: str,
    max_candidates: int
) -> List[Dict]:
    """为镜头搜索多样化的素材候选"""

    collected = []
    seen = set()
    queries = shot.get('search_queries') or ['cinematic background']

    # 1. 为每个查询词搜索素材
    for query_index, query in enumerate(queries[:3]):  # 使用前3个查询词
        query_candidates = []

        for provider in providers:
            try:
                results = provider.search(query, orientation=orientation, per_page=3)
                for candidate in results:
                    key = (candidate.get('provider'), candidate.get('source_id'))
                    if key in seen:
                        continue
                    seen.add(key)

                    # 标记素材来自哪个查询词
                    candidate['query_index'] = query_index
                    candidate['query'] = query
                    candidate['score'] = self._score_candidate(candidate, shot)
                    query_candidates.append(candidate)

            except Exception as e:
                logger.warning(f'素材搜索异常: {provider.provider_id}, {query}, {e}')

        # 2. 从每个查询词的结果中选择最佳素材
        query_candidates.sort(key=lambda x: x.get('score', 0), reverse=True)
        if query_candidates:
            # 每个查询词至少保留1个最佳素材
            collected.append(query_candidates[0])
            # 如果有多个高分素材，也加入
            for candidate in query_candidates[1:3]:
                if candidate['score'] >= query_candidates[0]['score'] * 0.8:
                    collected.append(candidate)

    # 3. 确保素材多样性
    collected = self._ensure_diversity(collected, shot)

    # 4. 按分数排序
    collected.sort(key=lambda item: item.get('score', 0), reverse=True)

    return collected[:max_candidates]

def _ensure_diversity(self, candidates: List[Dict], shot: Dict) -> List[Dict]:
    """确保素材多样性"""

    if len(candidates) <= 1:
        return candidates

    diverse_candidates = []
    used_providers = set()
    used_queries = set()

    # 优先选择不同Provider和不同查询词的素材
    for candidate in candidates:
        provider = candidate.get('provider')
        query_index = candidate.get('query_index')

        # 如果是新的Provider或新的查询词，优先加入
        if provider not in used_providers or query_index not in used_queries:
            diverse_candidates.append(candidate)
            used_providers.add(provider)
            used_queries.add(query_index)

            if len(diverse_candidates) >= 3:
                break

    # 如果不足3个，补充高分素材
    if len(diverse_candidates) < 3:
        for candidate in candidates:
            if candidate not in diverse_candidates:
                diverse_candidates.append(candidate)
                if len(diverse_candidates) >= 3:
                    break

    return diverse_candidates
```

**优点**:
- ✅ 每个镜头保证有1-3个不同风格的素材
- ✅ 避免所有素材来自同一个Provider
- ✅ 不同查询词带来不同视角的素材
- ✅ 用户有更多选择

#### B2. 智能素材评分优化

**核心思路**: 改进评分算法，考虑更多因素

```python
def _score_candidate(self, candidate: Dict, shot: Dict) -> int:
    """智能评分候选素材"""

    score = 40  # 基础分

    # 1. 时长匹配度 (0-25分)
    duration = float(candidate.get('duration') or 0.0)
    shot_duration = float(shot.get('duration') or 0.0)
    if duration >= shot_duration:
        score += 25
    elif duration >= shot_duration * 0.8:
        score += 20
    elif duration > 0:
        score += 10

    # 2. 分辨率质量 (0-20分)
    width = int(candidate.get('width') or 0)
    height = int(candidate.get('height') or 0)
    if width >= 1920 and height >= 1080:
        score += 20
    elif width >= 1280 and height >= 720:
        score += 15
    elif width >= 720:
        score += 10

    # 3. 横竖屏匹配 (0-10分)
    if width >= height:
        score += 10

    # 4. Provider优先级 (0-10分)
    provider_scores = {
        'wikimedia': 8,
        'nasa': 8,
        'videvo': 7,
        'videezy': 7,
        'mazwai': 9,  # 精选素材，高分
        'lifeofvids': 6,
        'splitshire': 6,
        'coverr': 7,
        'pexels': 5,
        'pixabay': 5
    }
    score += provider_scores.get(candidate.get('provider'), 5)

    # 5. 关键词匹配度 (0-15分) ✨新增
    keywords_match_score = self._calculate_keyword_match(candidate, shot)
    score += keywords_match_score

    # 6. 场景描述匹配度 (0-10分) ✨新增
    if shot.get('scene_description'):
        scene_match_score = self._calculate_scene_match(candidate, shot)
        score += scene_match_score

    return score

def _calculate_keyword_match(self, candidate: Dict, shot: Dict) -> int:
    """计算关键词匹配度"""

    shot_keywords = set(shot.get('keywords', []))
    candidate_query = candidate.get('query', '').lower()

    if not shot_keywords:
        return 0

    # 检查有多少关键词出现在候选素材的查询词中
    matches = sum(1 for kw in shot_keywords if kw.lower() in candidate_query)
    match_ratio = matches / len(shot_keywords)

    return int(match_ratio * 15)

def _calculate_scene_match(self, candidate: Dict, shot: Dict) -> int:
    """计算场景描述匹配度（需要AI或CLIP）"""

    # 简单版本：基于关键词
    scene_desc = shot.get('scene_description', '').lower()
    candidate_url = candidate.get('source_url', '').lower()

    # 检查场景关键词是否出现在素材URL中
    scene_keywords = re.findall(r'\w+', scene_desc)
    matches = sum(1 for kw in scene_keywords if kw in candidate_url)

    if matches > 0:
        return min(10, matches * 3)

    return 0
```
